```python
from sklearn.metrics.pairwise import cosine_similarity
import numpy as np

def collaborative_filter(user_id, user_item_matrix):
    # Calculate user similarity
    user_similarities = cosine_similarity(user_item_matrix)
    similar_users = user_similarities[user_id].argsort()[-10:][::-1]
    
    # Generate recommendations
    recommendations = {}
    for similar_user in similar_users:
        for item_id, rating in enumerate(user_item_matrix[similar_user]):
            if user_item_matrix[user_id][item_id] == 0 and rating > 0:
                recommendations[item_id] = recommendations.get(item_id, 0) + rating
    
    return sorted(recommendations.items(), key=lambda x: x[1], reverse=True)[:10]
```
