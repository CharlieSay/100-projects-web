<h2>Hint/Code snippet to start</h2>
```python
import face_recognition
import cv2

def encode_face(image_path):
    image = face_recognition.load_image_file(image_path)
    encodings = face_recognition.face_encodings(image)
    return encodings[0] if encodings else None

def recognize_face(unknown_image_path, known_encodings, known_names):
    unknown_encoding = encode_face(unknown_image_path)
    if unknown_encoding is None:
        return "No face found"
    
    matches = face_recognition.compare_faces(known_encodings, unknown_encoding)
    distances = face_recognition.face_distance(known_encodings, unknown_encoding)
    
    best_match_index = distances.argmin()
    if matches[best_match_index] and distances[best_match_index] < 0.6:
        return known_names[best_match_index]
    return "Unknown"
```
