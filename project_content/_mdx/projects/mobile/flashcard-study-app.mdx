```javascript
// Spaced repetition algorithm (SM-2)
function calculateNextReview(card, quality) {
  if (quality < 3) {
    card.repetitions = 0;
    card.interval = 1;
  } else {
    if (card.repetitions === 0) {
      card.interval = 1;
    } else if (card.repetitions === 1) {
      card.interval = 6;
    } else {
      card.interval = Math.round(card.interval * card.easinessFactor);
    }
    card.repetitions += 1;
  }
  card.easinessFactor = Math.max(1.3, card.easinessFactor + (0.1 - (5 - quality) * (0.08 + (5 - quality) * 0.02)));
  card.nextReview = new Date(Date.now() + card.interval * 24 * 60 * 60 * 1000);
}
```
